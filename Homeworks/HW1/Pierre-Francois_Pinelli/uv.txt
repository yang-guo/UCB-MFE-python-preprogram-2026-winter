# Create a virtual environment called HW1 with Python 3.12.9 (inside my submission folder)
uv venv HW1 --python 3.12.9
source HW1/bin/activate

# Install the packages from requirements.txt
uv pip install -r ../requirements.txt

# Downgrade requests to 2.24.0
uv pip install requests==2.24.0
uv pip show requests

# Upgrade requests to the latest version
uv pip install --upgrade requests
uv pip show requests

# Latest version of requests: 2.34.2
