# Install the exact Python version required by HW1.
uv python install 3.12.9

# Create the HW1 virtual environment with Python 3.12.9.
uv venv HW1 --python 3.12.9

# Package-management exercise completed on October 17, 2026.
# Install the official requirements in the homework environment (requests 2.25.0).
uv pip install --python .\HW1\Scripts\python.exe -r ..\requirements.txt
uv pip show --python .\HW1\Scripts\python.exe requests

# Downgrade requests to 2.24.0 and inspect its version without importing it.
uv pip install --python .\HW1\Scripts\python.exe requests==2.24.0
uv pip show --python .\HW1\Scripts\python.exe requests

# Upgrade requests to the latest available version.
uv pip install --python .\HW1\Scripts\python.exe --upgrade requests
uv pip show --python .\HW1\Scripts\python.exe requests

# Latest requests version after upgrading: 2.34.2.
# Verified on October 17, 2026, using the final uv pip show command above.
