uv venv --python 3.12.9 HW1
uv pip install --python HW1/bin/python -r ../requirements.txt
uv pip install --python HW1/bin/python requests==2.24.0
uv pip show --python HW1/bin/python requests
uv pip install --python HW1/bin/python --upgrade requests
uv pip show --python HW1/bin/python requests
HW1/bin/python -c "import requests; print(requests.__version__)"
HW1/bin/python game_of_life.py
